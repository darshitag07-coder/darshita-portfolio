C · PRACTICE PROJECT

C Calculator

A simple calculator made in C. This project helped me practise taking input, using operators, and displaying results.

See C code 
#include <stdio.h>

int main() {
    int num1, num2;
    char op;

    printf("Enter the first number: ");
    scanf("%d", &num1);

    printf("Enter the second number: ");
    scanf("%d", &num2);

    printf("Enter the operator (+, -, *, /, %%): ");
    scanf(" %c", &op);

    if (op == '+') {
        printf("Result: %d", num1 + num2);
    } else if (op == '-') {
        printf("Result: %d", num1 - num2);
    } else if (op == '*') {
        printf("Result: %d", num1 * num2);
    } else if (op == '%') {
        if (num2 != 0) {
            printf("Result: %d", num1 % num2);
        } else {
            printf("Cannot divide by zero.");
        }
    } else if (op == '/') {
        if (num2 != 0) {
            printf("Result: %d", num1 / num2);
        } else {
            printf("Cannot divide by zero.");
        }
    } else {
        printf("Invalid operator");
    }

    return 0;
}
PYTHON · PRACTICE PROJECT

Number Guessing Game

A simple game that chooses a random number from 1 to 100. The player keeps guessing until they find the correct number.

See Python code 
import random

number = random.randint(1, 100)
guess = 0

while guess != number:
    guess = int(input("Guess a number from 1 to 100: "))

    if guess < number:
        print("Too low")
    elif guess > number:
        print("Too high")

print("Correct!")
